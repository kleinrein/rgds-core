cmake_minimum_required(VERSION 3.16)
project(rgds_melonds_experiment LANGUAGES C CXX)
find_package(Threads REQUIRED)
if(NOT MELONDS_SOURCE)
    message(FATAL_ERROR "MELONDS_SOURCE must name the verified upstream checkout")
endif()
add_subdirectory("${MELONDS_SOURCE}" upstream)
add_executable(rgds-melonds-probe probe.cpp platform.cpp)
target_compile_features(rgds-melonds-probe PRIVATE cxx_std_17)
target_link_libraries(rgds-melonds-probe PRIVATE core Threads::Threads ${CMAKE_DL_LIBS})
target_link_options(rgds-melonds-probe PRIVATE -Wl,--strip-all)
set_property(TARGET rgds-melonds-probe PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)

if(RGDS_SDL_SDK)
    add_executable(melonds frontend.cpp platform.cpp)
    target_compile_features(melonds PRIVATE cxx_std_17)
    target_compile_definitions(melonds PRIVATE RGDS_MELONDS_FRONTEND)
    target_include_directories(melonds PRIVATE "${RGDS_SDL_SDK}/include/SDL2")
    target_link_libraries(melonds PRIVATE core Threads::Threads ${CMAKE_DL_LIBS} "${RGDS_SDL_SDK}/lib/libSDL2.so")
    target_link_options(melonds PRIVATE -Wl,--strip-all -Wl,--export-dynamic)
    set_property(TARGET melonds PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endif()
