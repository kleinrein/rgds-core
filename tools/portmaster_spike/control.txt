CUR_TTY=/dev/null

export controlfolder="/mnt/mmc/portmaster/PortMaster"
export directory="mnt/mmc/portmaster"

export ESUDO=""
export ESUDOKILL="-1"
export ESUDOKILL2="-1"

# Our launcher's pad mapping (runtime/emulators.py), not stock's (which swaps Select and Home).
RGDS_PAD="1900b655010000000100000000010000,ANBERNIC-rk3568-keys,a:b0,b:b1,x:b3,y:b2,back:b6,guide:b8,start:b7,leftstick:b9,rightstick:b12,leftshoulder:b4,rightshoulder:b5,dpup:h0.1,dpleft:h0.8,dpdown:h0.4,dpright:h0.2,leftx:a0,lefty:a1,rightx:a2,righty:a3,lefttrigger:b10,righttrigger:b11,platform:Linux,"
export SDL_GAMECONTROLLERCONFIG_FILE="$controlfolder/gamecontrollerdb.rgds.txt"
[ -f "$SDL_GAMECONTROLLERCONFIG_FILE" ] || echo "$RGDS_PAD" > "$SDL_GAMECONTROLLERCONFIG_FILE"

SDLDBFILE="${SDL_GAMECONTROLLERCONFIG_FILE}"
SDLDBUSERFILE="${SDL_GAMECONTROLLERCONFIG_FILE}"

get_controls() {
  LOWRES="N"
  ANALOGSTICKS=2
  DEVICE="1900b655010000000100000000010000"
  param_device="rgds"
  sdl_controllerconfig="$RGDS_PAD"
}

source $controlfolder/device_info.txt
source $controlfolder/funcs.txt

# The top panel only: device_info's DRM scan also sees DSI-2 and a writeback connector.
export DISPLAY_WIDTH=640 DISPLAY_HEIGHT=480

export GPTOKEYB2="$ESUDO env LD_PRELOAD=$controlfolder/libinterpose.${DEVICE_ARCH}.so $controlfolder/gptokeyb2 $ESUDOKILL2"
export GPTOKEYB="$ESUDO $controlfolder/gptokeyb $ESUDOKILL"
